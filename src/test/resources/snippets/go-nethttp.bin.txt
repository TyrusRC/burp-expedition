package main

import (
	"bytes"
	"encoding/base64"
	"fmt"
	"io"
	"net/http"
)

func main() {
	payload, _ := base64.StdEncoding.DecodeString("AAECf//+YQ==")
	req, err := http.NewRequest("PUT", "https://files.example.com:8443/up", bytes.NewReader(payload))
	if err != nil {
		panic(err)
	}
	req.Header.Add("Content-Type", "application/octet-stream")
	resp, err := http.DefaultClient.Do(req)
	if err != nil {
		panic(err)
	}
	defer resp.Body.Close()
	data, _ := io.ReadAll(resp.Body)
	fmt.Println(string(data))
}
