using System;
using System.Net.Http;
using System.Threading.Tasks;

class Program
{
    static async Task Main()
    {
        using var client = new HttpClient();
        var request = new HttpRequestMessage(new HttpMethod("PUT"), "https://files.example.com:8443/up");
        request.Content = new ByteArrayContent(Convert.FromBase64String("AAECf//+YQ=="));
        request.Content.Headers.Clear();
        request.Content.Headers.TryAddWithoutValidation("Content-Type", "application/octet-stream");
        var response = await client.SendAsync(request);
        Console.WriteLine(await response.Content.ReadAsStringAsync());
    }
}
