const axios = require("axios");

axios({
  method: "post",
  url: "https://api.example.com/v1/items?x=1&y=a%20b",
  headers: {
    "Content-Type": "application/json",
    "X-Token": "a\"b$c\\d",
    "Accept": "text/html, */*",
  },
  data: "{\"name\":\"it's $x\",\"t\":\"é\"}\n",
  responseType: "text",
  transformResponse: (d) => d,
})
  .then((res) => console.log(res.data))
  .catch((err) => console.error(err));
